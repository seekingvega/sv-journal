import backtrader as bt
from datetime import datetime

class PercentSizer(bt.Sizer):
    params = (
        ('percents', 1),  # Percentage of portfolio to risk
    )

    def _getsizing(self, comminfo, cash, data, isbuy):
        if isbuy:
            position = cash * (self.params.percents / 100)
            return position / data[0]
        return self.broker.getposition(data).size
    
class MultiTimeframeMACD(bt.Strategy):
    params = (
        ('fast_period', 12),
        ('slow_period', 26),
        ('signal_period', 9),
        ('atr_period', 14),
        ('atr_multiplier', 2),
    )

    def __init__(self):
        # Get the data feeds by name
        self.weekly = self.getdatabyname('weekly')
        self.monthly = self.getdatabyname('monthly')
        self.daily = self.getdatabyname('daily')

        # Monthly MACD
        self.monthly_macd = bt.indicators.MACD(
            self.monthly,
            period_me1=self.p.fast_period,
            period_me2=self.p.slow_period,
            period_signal=self.p.signal_period
        )
        
        # Weekly MACD
        self.weekly_macd = bt.indicators.MACD(
            self.weekly,
            period_me1=self.p.fast_period,
            period_me2=self.p.slow_period,
            period_signal=self.p.signal_period
        )
        
        # Weekly MACD crossover
        self.weekly_crossover = bt.indicators.CrossOver(
            self.weekly_macd.macd, 
            self.weekly_macd.signal
        )

        # ATR for stop-loss calculation
        self.atr = bt.indicators.ATR(
            self.daily,
            period=self.p.atr_period
        )

        # Keep track of stop loss
        self.stop_loss = None

    def notify_order(self, order):
        if order.status in [order.Submitted, order.Accepted]:
            return

        if order.status in [order.Completed]:
            if order.isbuy():
                self.stop_loss = order.executed.price - self.atr[0] * self.p.atr_multiplier
            elif order.issell():
                self.stop_loss = None

        elif order.status in [order.Canceled, order.Margin, order.Rejected]:
            print(f'Order Canceled/Margin/Rejected: {order.status}')

    def next(self):
        # Check if we should exit based on stop loss
        if self.position and self.daily.close[0] < self.stop_loss:
            self.close()
            return

        if not self.position:  # If we're not in the market
            if self.monthly_macd.macd > 0 and self.weekly_crossover > 0:
                # Enter long if monthly MACD is positive and weekly MACD just turned positive
                self.buy()
        
        elif self.weekly_crossover < 0:  # If we're in the market and weekly MACD turns negative
            self.close()  # Exit the position

Running the Backtest

Code 
def run_strategy(do_plot: bool = False):
    cerebro = bt.Cerebro()

    # Add data feeds
    data = bt.feeds.PandasData(dataname = gld)
    cerebro.adddata(data, name="daily")
    
    # Resample to weekly timeframe
    cerebro.resampledata(data, timeframe=bt.TimeFrame.Weeks, name="weekly")
    
    # Resample to monthly timeframe
    cerebro.resampledata(data, timeframe=bt.TimeFrame.Months, name="monthly")

    # Add strategy
    cerebro.addstrategy(MultiTimeframeMACD)

    # Set initial cash
    cerebro.broker.setcash(100000.0)

    # Run the strategy
    result = cerebro.run()

    # Print final portfolio value
    print(f'Final Portfolio Value: ${cerebro.broker.getvalue():.2f}')

    if do_plot:
        cerebro.plot()
    return result
